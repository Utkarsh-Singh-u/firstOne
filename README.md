#include <stdio.h>
#include <stdlib.h>

void bresenham(int x1, int y1, int x2, int y2)
{
    int dx = abs(x2 - x1);
    int dy = abs(y2 - y1);

    int sx = (x2 >= x1) ? 1 : -1;
    int sy = (y2 >= y1) ? 1 : -1;

    int x = x1;
    int y = y1;

    printf("\nPoints generated by Bresenham's Algorithm:\n");
    printf("(%d, %d)\n", x, y);

    // For slope <= 1
    if (dx >= dy)
    {
        int p = 2 * dy - dx;

        while (x != x2)
        {
            x += sx;

            if (p < 0)
            {
                p = p + 2 * dy;
            }
            else
            {
                y += sy;
                p = p + 2 * (dy - dx);
            }

            printf("(%d, %d)\n", x, y);
        }
    }

    // For slope > 1
    else
    {
        int p = 2 * dx - dy;

        while (y != y2)
        {
            y += sy;

            if (p < 0)
            {
                p = p + 2 * dx;
            }
            else
            {
                x += sx;
                p = p + 2 * (dx - dy);
            }

            printf("(%d, %d)\n", x, y);
        }
    }
}

int main()
{
    int x1, y1, x2, y2;

    printf("Enter first point (x1, y1): ");
    scanf("%d %d", &x1, &y1);

    printf("Enter second point (x2, y2): ");
    scanf("%d %d", &x2, &y2);

    bresenham(x1, y1, x2, y2);

    return 0;
}
